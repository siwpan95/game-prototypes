class Vector2:
    """二维向量类"""
    def __init__(self, x: float, y: float):
        self.x = x
        self.y = y
    
    def rotate(self, angle: float) -> 'Vector2':
        """旋转向量"""
        cos_a = math.cos(angle)
        sin_a = math.sin(angle)
        return Vector2(
            self.x * cos_a - self.y * sin_a,
            self.x * sin_a + self.y * cos_a
        )
    
    def translate(self, dx: float, dy: float) -> 'Vector2':
        """平移向量"""
        return Vector2(self.x + dx, self.y + dy)
    
    def distance_to(self, other: 'Vector2') -> float:
        """计算到另一个点的距离"""
        return math.sqrt((self.x - other.x)**2 + (self.y - other.y)**2)
    
    def copy(self) -> 'Vector2':
        """复制向量"""
        return Vector2(self.x, self.y)
    
    def to_tuple(self) -> Tuple[float, float]:
        """转换为元组"""
        return (self.x, self.y)

class WallSegment:
    """墙段类"""
    def __init__(self, start: Vector2, end: Vector2, color: Tuple[int, int, int]):
        self.start = start
        self.end = end
        self.color = color
    
    def translate(self, dx: float, dy: float) -> 'WallSegment':
        """平移墙段"""
        return WallSegment(
            self.start.translate(dx, dy),
            self.end.translate(dx, dy),
            self.color
        )
    
    def rotate(self, angle: float, center: Vector2) -> 'WallSegment':
        """绕中心点旋转墙段"""
        # 相对于中心点的坐标
        start_rel = Vector2(
            self.start.x - center.x,
            self.start.y - center.y
        )
        end_rel = Vector2(
            self.end.x - center.x,
            self.end.y - center.y
        )
        
        # 旋转
        start_rot = start_rel.rotate(angle)
        end_rot = end_rel.rotate(angle)
        
        # 转换回绝对坐标
        return WallSegment(
            Vector2(start_rot.x + center.x, start_rot.y + center.y),
            Vector2(end_rot.x + center.x, end_rot.y + center.y),
            self.color
        )

class Player:
    """玩家类"""
    def __init__(self, x: float, y: float, angle: float = 0.0):
        self.pos = Vector2(x, y)
        self.angle = angle  # 角度，弧度制
        self.fov = math.pi / 2  # 90度视野
        self.move_speed = 0.1
        self.rotate_speed = 0.05
        self.height = 0.5  # 玩家眼睛高度（相对于墙高）
        
    def move_forward(self, distance: float):
        """向前移动"""
        dx = math.cos(self.angle) * distance
        dy = math.sin(self.angle) * distance
        self.pos.x += dx
        self.pos.y += dy
    
    def move_backward(self, distance: float):
        """向后移动"""
        dx = math.cos(self.angle) * distance
        dy = math.sin(self.angle) * distance
        self.pos.x -= dx
        self.pos.y -= dy
    
    def rotate_left(self, angle: float):
        """向左旋转"""
        self.angle -= angle
    
    def rotate_right(self, angle: float):
        """向右旋转"""
        self.angle += angle
    
    def get_direction_vector(self) -> Vector2:
        """获取方向向量"""
        return Vector2(math.cos(self.angle), math.sin(self.angle))

class Renderer:
    """3D渲染器"""
    def __init__(self, screen_width: int, screen_height: int):
        self.screen_width = screen_width
        self.screen_height = screen_height
        self.half_height = screen_height // 2
        self.projection_plane_dist = 1.0  # 投影平面距离
        self.max_render_distance = 20.0  # 最大渲染距离
        
    def project_point(self, point: Vector2, player: Player) -> Optional[Tuple[float, float, float]]:
        """
        将3D点投影到2D屏幕
        
        返回: (屏幕X坐标, 屏幕Y顶部, 屏幕Y底部) 或 None（如果点在视野外）
        """
        # 1. 转换到玩家相对坐标系
        rel_x = point.x - player.pos.x
        rel_y = point.y - player.pos.y
        
        # 2. 旋转世界，使玩家朝向正上方
        # 计算相对于玩家朝向的坐标
        cos_a = math.cos(-player.angle)
        sin_a = math.sin(-player.angle)
        
        # 旋转坐标
        rotated_x = rel_x * cos_a - rel_y * sin_a
        rotated_y = rel_x * sin_a + rel_y * cos_a
        
        # 3. 检查点是否在视野前方
        if rotated_y <= 0.001:  # 避免除以零和负值
            return None
        
        # 4. 透视投影：X坐标除以距离
        proj_x = rotated_x / rotated_y
        
        # 5. 检查点是否在视野内（FOV检查）
        if abs(proj_x) > math.tan(player.fov / 2):
            return None
        
        # 6. 将投影坐标映射到屏幕
        screen_x = (proj_x / math.tan(player.fov / 2) + 1) * (self.screen_width / 2)
        
        # 7. 计算墙的高度（基于距离）
        wall_height = 1.0  # 标准墙高
        wall_top = self.half_height - (wall_height / rotated_y) * (self.screen_height / 2)
        wall_bottom = self.half_height + (wall_height / rotated_y) * (self.screen_height / 2)
        
        return screen_x, wall_top, wall_bottom, rotated_y
    
    def render_wall_segment(self, wall: WallSegment, player: Player):
        """渲染一个墙段"""
        # 获取墙段两端的投影
        start_proj = self.project_point(wall.start, player)
        end_proj = self.project_point(wall.end, player)
        
        if start_proj is None or end_proj is None:
            return
        
        screen_x1, wall_top1, wall_bottom1, dist1 = start_proj
        screen_x2, wall_top2, wall_bottom2, dist2 = end_proj
        
        # 确保x1 < x2
        if screen_x1 > screen_x2:
            screen_x1, screen_x2 = screen_x2, screen_x1
            wall_top1, wall_top2 = wall_top2, wall_top1
            wall_bottom1, wall_bottom2 = wall_bottom2, wall_bottom1
            dist1, dist2 = dist2, dist1
        
        # 限制在屏幕范围内
        if screen_x1 >= self.screen_width or screen_x2 <= 0:
            return
        
        # 裁剪到屏幕
        if screen_x1 < 0:
            screen_x1 = 0
        if screen_x2 > self.screen_width:
            screen_x2 = self.screen_width
        
        # 根据距离调整颜色亮度
        avg_dist = (dist1 + dist2) / 2
        brightness = max(50, 255 - int(avg_dist * 20))
        color = (
            min(255, wall.color[0] * brightness // 255),
            min(255, wall.color[1] * brightness // 255),
            min(255, wall.color[2] * brightness // 255)
        )
        
        # 绘制墙段
        for x in range(int(screen_x1), int(screen_x2)):
            # 线性插值计算当前x对应的墙顶和墙底
            t = (x - screen_x1) / (screen_x2 - screen_x1) if screen_x2 != screen_x1 else 0
            current_top = wall_top1 + (wall_top2 - wall_top1) * t
            current_bottom = wall_bottom1 + (wall_bottom2 - wall_bottom1) * t
            
            # 绘制垂直线段
            pygame.draw.line(
                screen, 
                color, 
                (x, int(current_top)), 
                (x, int(current_bottom)), 
                1
            )

def create_room(width: float, height: float, center_x: float, center_y: float) -> List[WallSegment]:
    """创建一个矩形房间"""
    walls = []
    
    # 房间颜色
    colors = [
        (200, 100, 100),  # 红
        (100, 200, 100),  # 绿
        (100, 100, 200),  # 蓝
        (200, 200, 100),  # 黄
    ]
    
    # 定义四个角
    corners = [
        Vector2(center_x - width/2, center_y - height/2),
        Vector2(center_x + width/2, center_y - height/2),
        Vector2(center_x + width/2, center_y + height/2),
        Vector2(center_x - width/2, center_y + height/2),
    ]
    
    # 创建四面墙
    for i in range(4):
        start = corners[i]
        end = corners[(i + 1) % 4]
        walls.append(WallSegment(start, end, colors[i]))
    
    return walls

def create_complex_scene() -> List[WallSegment]:
    """创建更复杂的场景"""
    walls = []
    
    # 主房间
    main_room = create_room(8, 6, 0, 0)
    walls.extend(main_room)
    
    # 内部隔断
    walls.append(WallSegment(Vector2(-2, -1), Vector2(-2, 2), (150, 150, 200)))
    walls.append(WallSegment(Vector2(2, -2), Vector2(2, 1), (200, 150, 150)))
    
    # 外部走廊
    walls.append(WallSegment(Vector2(-4, -4), Vector2(4, -4), (100, 200, 200)))
    walls.append(WallSegment(Vector2(4, -4), Vector2(4, 4), (200, 100, 200)))
    walls.append(WallSegment(Vector2(4, 4), Vector2(-4, 4), (200, 200, 100)))
    walls.append(WallSegment(Vector2(-4, 4), Vector2(-4, -4), (100, 100, 100)))
    
    return walls

def draw_2d_map(surface, walls: List[WallSegment], player: Player, scale: float = 20, offset_x: float = 50, offset_y: float = 50):
    """绘制2D地图"""
    # 绘制墙壁
    for wall in walls:
        start_x = int(wall.start.x * scale + offset_x)
        start_y = int(wall.start.y * scale + offset_y)
        end_x = int(wall.end.x * scale + offset_x)
        end_y = int(wall.end.y * scale + offset_y)
        
        pygame.draw.line(surface, wall.color, (start_x, start_y), (end_x, end_y), 2)
    
    # 绘制玩家位置
    player_x = int(player.pos.x * scale + offset_x)
    player_y = int(player.pos.y * scale + offset_y)
    pygame.draw.circle(surface, (255, 255, 0), (player_x, player_y), 5)
    
    # 绘制玩家方向
    dir_length = 15
    dir_x = player_x + math.cos(player.angle) * dir_length
    dir_y = player_y + math.sin(player.angle) * dir_length
    pygame.draw.line(surface, (255, 255, 0), (player_x, player_y), (int(dir_x), int(dir_y)), 2)
    
    # 绘制玩家视野
    fov = player.fov
    for angle_offset in [-fov/2, fov/2]:
        ray_angle = player.angle + angle_offset
        ray_end_x = player_x + math.cos(ray_angle) * dir_length * 1.5
        ray_end_y = player_y + math.sin(ray_angle) * dir_length * 1.5
        pygame.draw.line(surface, (255, 200, 0, 128), (player_x, player_y), (int(ray_end_x), int(ray_end_y)), 1)
    
    # 绘制坐标轴
    pygame.draw.line(surface, (100, 100, 100), (offset_x, offset_y), (offset_x + 100, offset_y), 1)
    pygame.draw.line(surface, (100, 100, 100), (offset_x, offset_y), (offset_x, offset_y + 100), 1)

def main():
    """主函数"""
    # 创建玩家
    player = Player(0, 0, 0)
    
    # 创建渲染器
    renderer = Renderer(SCREEN_WIDTH, SCREEN_HEIGHT)
    
    # 创建场景
    walls = create_complex_scene()
    
    # 添加一些随机墙
    for _ in range(5):
        x1 = random.uniform(-6, 6)
        y1 = random.uniform(-6, 6)
        x2 = x1 + random.uniform(-3, 3)
        y2 = y1 + random.uniform(-3, 3)
        color = (random.randint(100, 200), random.randint(100, 200), random.randint(100, 200))
        walls.append(WallSegment(Vector2(x1, y1), Vector2(x2, y2), color))
    
    # 游戏主循环
    running = True
    show_map = True
    show_info = True
    
    while running:
        # 处理事件
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_m:
                    show_map = not show_map
                elif event.key == pygame.K_i:
                    show_info = not show_info
                elif event.key == pygame.K_r:
                    # 重置玩家位置
                    player = Player(0, 0, 0)
        
        # 处理键盘输入
        keys = pygame.key.get_pressed()
        if keys[pygame.K_w] or keys[pygame.K_UP]:
            player.move_forward(player.move_speed)
        if keys[pygame.K_s] or keys[pygame.K_DOWN]:
            player.move_backward(player.move_speed)
        if keys[pygame.K_a]:
            player.rotate_left(player.rotate_speed)
        if keys[pygame.K_d]:
            player.rotate_right(player.rotate_speed)
        if keys[pygame.K_LEFT]:
            player.rotate_left(player.rotate_speed * 0.5)
        if keys[pygame.K_RIGHT]:
            player.rotate_right(player.rotate_speed * 0.5)
        
        # 绘制3D视图
        screen.fill((30, 30, 40))  # 深蓝色背景
        
        # 绘制地板和天花板
        pygame.draw.rect(screen, (40, 40, 60), (0, 0, SCREEN_WIDTH, HALF_HEIGHT))  # 天花板
        pygame.draw.rect(screen, (50, 40, 30), (0, HALF_HEIGHT, SCREEN_WIDTH, HALF_HEIGHT))  # 地板
        
        # 渲染所有墙壁
        for wall in walls:
            renderer.render_wall_segment(wall, player)
        
        # 绘制2D地图
        if show_map:
            map_surface = pygame.Surface((300, 300), pygame.SRCALPHA)
            map_surface.fill((0, 0, 0, 150))  # 半透明黑色背景
            draw_2d_map(map_surface, walls, player, scale=20, offset_x=150, offset_y=150)
            screen.blit(map_surface, (SCREEN_WIDTH - 320, 20))
        
        # 显示信息
        if show_info:
            info_text = [
                f"位置: ({player.pos.x:.2f}, {player.pos.y:.2f})",
                f"方向: {math.degrees(player.angle):.1f}°",
                f"控制: W/A/S/D - 移动, 方向键 - 旋转",
                f"M - 切换地图, I - 切换信息, R - 重置",
                f"墙段数量: {len(walls)}",
                f"FPS: {int(clock.get_fps())}",
                "",
                "伪3D渲染原理:",
                "1. 将世界旋转到玩家视角",
                "2. 投影到相机平面",
                "3. 透视投影: x' = x / z",
                "4. 根据距离调整墙高",
                "5. 绘制到屏幕"
            ]
            
            for i, text in enumerate(info_text):
                text_surface = font.render(text, True, (255, 255, 200))
                screen.blit(text_surface, (20, 20 + i * 25))
        
        # 更新显示
        pygame.display.flip()
        clock.tick(60)
    
    pygame.quit()

if __name__ == "__main__":

实现原理说明
这个伪3D渲染引擎基于以下核心原理：
1. 坐标变换
* 将世界坐标转换为相对于玩家的坐标系
* 旋转世界使玩家朝向正上方，简化后续计算
2. 透视投影
* 使用公式 x' = x / z 将3D坐标投影到2D屏幕
* 靠近相机的物体看起来更大，远处的物体看起来更小
3. 视野裁剪
* 只渲染玩家视野内（FOV范围内）的物体
* 忽略相机后面的物体（z ≤ 0）
4. 墙高计算
* 墙的高度与距离成反比：墙高 = 基准高度 / 距离
* 距离越远，墙在屏幕上看起来越矮
5. 颜色渐变
* 根据距离调整颜色亮度，模拟光照衰减
* 远处的墙颜色更暗，增加深度感
控制说明
* W/S 或 上/下箭头：前进/后退
* A/D 或 左/右箭头：旋转视角
* M：切换2D地图显示
* I：切换信息显示
* R：重置玩家位置
扩展建议
1. 纹理映射：为墙壁添加纹理而不是纯色
2. 光照模型：实现更复杂的光照和阴影
3. 实体渲染：添加可交互的3D物体
4. 碰撞检测：防止玩家穿墙
5. 天空盒和地板纹理：增强沉浸感
这个实现展示了早期3D游戏（如《毁灭战士》）的核心渲染原理，通过简单的几何投影创建3D视觉效果，而无需真正的3D图形硬件。

class DopplerEffect:
    """多普勒效应模拟 - 用于距离感知和运动感知"""
    
    @staticmethod
    def calculate_frequency_shift(distance: float, velocity: float, 
                                 base_frequency: float = 440.0) -> float:
        """
        计算多普勒频移
        distance: 距离
        velocity: 相对速度（正表示靠近，负表示远离）
        base_frequency: 基础频率
        返回：感知频率
        """
        # 简化多普勒公式：f' = f * (c / (c ± v))
        speed_of_sound = 343.0  # 声速 m/s
        
        if velocity > 0:  # 靠近
            return base_frequency * (speed_of_sound / (speed_of_sound - velocity))
        else:  # 远离
            return base_frequency * (speed_of_sound / (speed_of_sound + abs(velocity)))
    
    @staticmethod
    def distance_to_color(distance: float, max_distance: float = 20.0) -> Tuple[int, int, int]:
        """
        将距离转换为颜色（模拟多普勒频移的视觉表现）
        距离越近，颜色越暖（红/黄），距离越远，颜色越冷（蓝/紫）
        """
        if distance <= 0:
            distance = 0.001
            
        # 归一化距离
        t = min(distance / max_distance, 1.0)
        
        # 使用色彩渐变模拟多普勒效应
        if t < 0.3:  # 近距离 - 暖色
            r = 255
            g = int(200 * (1 - t/0.3))
            b = 50
        elif t < 0.6:  # 中距离
            r = int(255 * (1 - (t-0.3)/0.3))
            g = 150
            b = 100
        else:  # 远距离 - 冷色
            r = 50
            g = 100
            b = int(255 * (t-0.6)/0.4)
            
        return (r, g, b)
    
    @staticmethod
    def apply_motion_blur(surface, velocity: float, angle: float):
        """
        应用运动模糊模拟多普勒效应
        速度越快，模糊越明显
        """
        if abs(velocity) < 0.1:
            return surface
            
        # 创建模糊表面
        blur_surface = pygame.Surface((SCREEN_WIDTH, SCREEN_HEIGHT), pygame.SRCALPHA)
        
        # 根据速度和方向计算模糊强度
        blur_intensity = int(min(abs(velocity) * 5, 10))
        
        for i in range(blur_intensity):
            offset_x = math.cos(angle) * i
            offset_y = math.sin(angle) * i
            alpha = 100 // (i + 1)  # 递减的透明度
            
            # 绘制模糊副本
            blur_surface.blit(surface, (offset_x, offset_y))
            blur_surface.set_alpha(alpha)
            screen.blit(blur_surface, (0, 0))
            
        return surface

class NoiseAlgorithm:
    """噪声算法 - 用于生成自然纹理和视觉效果"""
    
    @staticmethod
    def perlin_noise_1d(x: float, seed: int = 0) -> float:
        """简化的一维Perlin噪声"""
        x0 = int(x)
        x1 = x0 + 1
        dx = x - x0
        
        # 使用随机梯度
        random.seed(seed + x0)
        g0 = random.uniform(-1, 1)
        random.seed(seed + x1)
        g1 = random.uniform(-1, 1)
        
        # 插值
        n0 = g0 * dx
        n1 = g1 * (dx - 1)
        
        # 平滑插值
        fade = dx * dx * (3 - 2 * dx)
        return n0 + fade * (n1 - n0)
    
    @staticmethod
    def fractal_noise(x: float, octaves: int = 4, persistence: float = 0.5) -> float:
        """分形噪声（多重八度Perlin噪声）"""
        value = 0.0
        amplitude = 1.0
        frequency = 1.0
        
        for _ in range(octaves):
            value += NoiseAlgorithm.perlin_noise_1d(x * frequency) * amplitude
            amplitude *= persistence
            frequency *= 2.0
            
        return value
    
    @staticmethod
    def generate_wall_texture(width: int, height: int, seed: int = 0) -> pygame.Surface:
        """生成墙面纹理"""
        surface = pygame.Surface((width, height))
        
        for x in range(width):
            for y in range(height):
                # 使用噪声生成颜色变化
                noise_val = NoiseAlgorithm.fractal_noise(x * 0.1 + y * 0.05 + seed * 100)
                
                # 将噪声值转换为灰度
                gray = int((noise_val + 1) * 128)
                
                # 添加一些颜色变化
                r = min(255, gray + random.randint(-20, 20))
                g = min(255, gray + random.randint(-20, 20))
                b = min(255, gray + random.randint(-20, 20))
                
                surface.set_at((x, y), (r, g, b))
                
        return surface
    
    @staticmethod
    def apply_surface_noise(surface, intensity: float = 0.1):
        """为表面添加噪声效果"""
        noise_surface = surface.copy()
        for x in range(surface.get_width()):
            for y in range(surface.get_height()):
                r, g, b, a = surface.get_at((x, y))
                
                # 添加随机噪声
                noise = random.uniform(-intensity, intensity)
                r = int(min(255, max(0, r + r * noise)))
                g = int(min(255, max(0, g + g * noise)))
                b = int(min(255, max(0, b + b * noise)))
                
                noise_surface.set_at((x, y), (r, g, b, a))
                
        return noise_surface

class DOOMRenderer:
    """DOOM风格渲染器 - 使用射线投射算法"""
    
    def __init__(self, screen_width: int, screen_height: int):
        self.screen_width = screen_width
        self.screen_height = screen_height
        self.half_height = screen_height // 2
        self.fov = math.pi / 2  # 90度视野
        self.projection_plane = 0.5  # 投影平面距离
        
        # 预计算射线角度
        self.ray_angles = []
        self.num_rays = screen_width // 2
        self.setup_rays()
        
        # 纹理缓存
        self.textures = {}
        self.load_textures()
    
    def setup_rays(self):
        """预计算所有射线的角度"""
        for i in range(self.num_rays):
            ray_angle = (i / self.num_rays - 0.5) * self.fov
            self.ray_angles.append(ray_angle)
    
    def load_textures(self):
        """加载纹理"""
        # 创建一些基本纹理
        self.textures['brick'] = NoiseAlgorithm.generate_wall_texture(64, 64, seed=1)
        self.textures['stone'] = NoiseAlgorithm.generate_wall_texture(64, 64, seed=2)
        self.textures['metal'] = NoiseAlgorithm.generate_wall_texture(64, 64, seed=3)
        self.textures['wood'] = NoiseAlgorithm.generate_wall_texture(64, 64, seed=4)
    
    def cast_ray(self, start_x: float, start_y: float, angle: float, 
                walls: List['Wall']) -> Tuple[float, Optional['Wall']]:
        """
        投射一条射线，返回击中墙壁的距离和墙壁对象
        使用DDA算法（数字微分分析）
        """
        # 射线方向
        ray_dir_x = math.cos(angle)
        ray_dir_y = math.sin(angle)
        
        # 玩家所在的网格坐标
        map_x = int(start_x)
        map_y = int(start_y)
        
        # 射线长度
        ray_length = 0
        max_distance = 20.0
        
        # DDA算法
        delta_dist_x = abs(1 / ray_dir_x) if ray_dir_x != 0 else 1e30
        delta_dist_y = abs(1 / ray_dir_y) if ray_dir_y != 0 else 1e30
        
        # 步进方向
        step_x = 1 if ray_dir_x >= 0 else -1
        step_y = 1 if ray_dir_y >= 0 else -1
        
        # 到下一个网格边界的距离
        if ray_dir_x < 0:
            side_dist_x = (start_x - map_x) * delta_dist_x
        else:
            side_dist_x = (map_x + 1.0 - start_x) * delta_dist_x
            
        if ray_dir_y < 0:
            side_dist_y = (start_y - map_y) * delta_dist_y
        else:
            side_dist_y = (map_y + 1.0 - start_y) * delta_dist_y
        
        # 碰撞检测
        hit = False
        side = 0  # 0=x面, 1=y面
        
        while not hit and ray_length < max_distance:
            if side_dist_x < side_dist_y:
                side_dist_x += delta_dist_x
                map_x += step_x
                side = 0
            else:
                side_dist_y += delta_dist_y
                map_y += step_y
                side = 1
            
            # 检查是否击中墙壁
            for wall in walls:
                if int(wall.x) == map_x and int(wall.y) == map_y:
                    hit = True
                    break
            
            ray_length = side_dist_x if side == 0 else side_dist_y
        
        if hit:
            # 计算精确距离
            if side == 0:
                distance = (map_x - start_x + (1 - step_x) / 2) / ray_dir_x
            else:
                distance = (map_y - start_y + (1 - step_y) / 2) / ray_dir_y
                
            return distance, wall
        
        return max_distance, None
    
    def render_column(self, x: int, distance: float, wall, side: int, 
                     player_angle: float, ray_angle: float):
        """
        渲染一列墙
        x: 屏幕X坐标
        distance: 到墙的距离
        wall: 墙壁对象
        side: 击中的面（0=垂直，1=水平）
        """
        if distance <= 0:
            return
            
        # 计算墙的高度
        wall_height = int(self.screen_height / distance)
        
        # 计算墙的顶部和底部
        wall_top = self.half_height - wall_height // 2
        wall_bottom = wall_top + wall_height
        
        # 限制在屏幕范围内
        wall_top = max(0, wall_top)
        wall_bottom = min(self.screen_height, wall_bottom)
        
        if wall_top >= wall_bottom:
            return
        
        # 计算纹理坐标
        if side == 0:  # 垂直面
            wall_x = wall.y + distance * math.sin(ray_angle)
        else:  # 水平面
            wall_x = wall.x + distance * math.cos(ray_angle)
        
        tex_x = int((wall_x - int(wall_x)) * 64) % 64
        
        # 获取纹理
        texture = self.textures.get(wall.texture_type, self.textures['brick'])
        
        # 渲染纹理列
        for y in range(wall_top, wall_bottom):
            tex_y = int((y - wall_top) * 64 / wall_height) % 64
            
            # 获取纹理颜色
            color = texture.get_at((tex_x, tex_y))
            
            # 根据距离调整亮度
            brightness = max(0.2, 1.0 - distance / 20.0)
            r, g, b = color[:3]
            r = int(r * brightness)
            g = int(g * brightness)
            b = int(b * brightness)
            
            # 应用多普勒效应颜色
            doppler_color = DopplerEffect.distance_to_color(distance)
            
            # 混合颜色
            mix_factor = 0.7
            final_r = int(r * (1 - mix_factor) + doppler_color[0] * mix_factor)
            final_g = int(g * (1 - mix_factor) + doppler_color[1] * mix_factor)
            final_b = int(b * (1 - mix_factor) + doppler_color[2] * mix_factor)
            
            # 绘制像素
            screen.set_at((x, y), (final_r, final_g, final_b))
    
    def render(self, player, walls: List['Wall']):
        """渲染整个场景"""
        # 绘制天空和地面
        pygame.draw.rect(screen, (100, 150, 255), (0, 0, SCREEN_WIDTH, HALF_HEIGHT))  # 天空
        pygame.draw.rect(screen, (50, 100, 50), (0, HALF_HEIGHT, SCREEN_WIDTH, HALF_HEIGHT))  # 地面
        
        # 投射射线
        for i, ray_angle in enumerate(self.ray_angles):
            # 计算射线在世界中的角度
            world_angle = player.angle + ray_angle
            
            # 投射射线
            distance, wall = self.cast_ray(
                player.x, player.y, world_angle, walls
            )
            
            if wall:
                # 计算击中的面
                side = 0 if abs(math.sin(world_angle)) > abs(math.cos(world_angle)) else 1
                
                # 渲染这一列
                x = i * 2
                self.render_column(x, distance, wall, side, player.angle, world_angle)
                self.render_column(x + 1, distance, wall, side, player.angle, world_angle)

class Vector2:
    """二维向量"""
    def __init__(self, x: float, y: float):
        self.x = x
        self.y = y
    
    def distance_to(self, other: 'Vector2') -> float:
        """计算到另一个点的距离"""
        return math.sqrt((self.x - other.x) ** 2 + (self.y - other.y) ** 2)
    
    def __str__(self):
        return f"({self.x:.2f}, {self.y:.2f})"

class Wall:
    """墙壁"""
    def __init__(self, x: float, y: float, texture_type: str = "brick"):
        self.x = x
        self.y = y
        self.texture_type = texture_type

class Player:
    """玩家"""
    def __init__(self, x: float, y: float, angle: float = 0.0):
        self.x = x
        self.y = y
        self.angle = angle
        self.velocity_x = 0.0
        self.velocity_y = 0.0
        self.speed = 0.1
        self.rotate_speed = 0.05
        self.height = 0.5
    
    def move(self, forward: float, strafe: float):
        """移动玩家"""
        # 计算移动方向
        move_x = math.cos(self.angle) * forward + math.cos(self.angle + math.pi/2) * strafe
        move_y = math.sin(self.angle) * forward + math.sin(self.angle + math.pi/2) * strafe
        
        # 更新位置
        self.x += move_x * self.speed
        self.y += move_y * self.speed
        
        # 更新速度（用于多普勒效应）
        self.velocity_x = move_x * self.speed
        self.velocity_y = move_y * self.speed
    
    def rotate(self, direction: float):
        """旋转玩家"""
        self.angle += direction * self.rotate_speed
        
        # 归一化角度
        if self.angle > 2 * math.pi:
            self.angle -= 2 * math.pi
        elif self.angle < 0:
            self.angle += 2 * math.pi
    
    def get_speed(self) -> float:
        """获取速度大小"""
        return math.sqrt(self.velocity_x ** 2 + self.velocity_y ** 2)

class GameWorld:
    """游戏世界"""
    def __init__(self):
        self.player = Player(5.5, 5.5, 0.0)
        self.walls = []
        self.generate_maze(10, 10)
        
        # 渲染器
        self.renderer = DOOMRenderer(SCREEN_WIDTH, SCREEN_HEIGHT)
        self.render_mode = RenderMode.DOOM_STYLE
        
        # 性能统计
        self.frame_count = 0
        self.fps = 0
        self.last_time = pygame.time.get_ticks()
    
    def generate_maze(self, width: int, height: int):
        """生成迷宫"""
        # 创建外墙
        for x in range(width):
            self.walls.append(Wall(x, 0, "stone"))
            self.walls.append(Wall(x, height - 1, "stone"))
        
        for y in range(height):
            self.walls.append(Wall(0, y, "stone"))
            self.walls.append(Wall(width - 1, y, "stone"))
        
        # 创建内墙
        for x in range(2, width - 2, 2):
            for y in range(2, height - 2, 2):
                # 随机选择纹理
                textures = ["brick", "stone", "metal", "wood"]
                texture = random.choice(textures)
                
                # 随机决定是否创建墙
                if random.random() > 0.3:
                    # 随机方向
                    if random.random() > 0.5:
                        self.walls.append(Wall(x, y, texture))
                        if random.random() > 0.5:
                            self.walls.append(Wall(x + 1, y, texture))
                    else:
                        self.walls.append(Wall(x, y, texture))
                        if random.random() > 0.5:
                            self.walls.append(Wall(x, y + 1, texture))
    
    def update(self):
        """更新游戏状态"""
        keys = pygame.key.get_pressed()
        
        # 移动
        forward = 0.0
        strafe = 0.0
        
        if keys[pygame.K_w]:
            forward += 1.0
        if keys[pygame.K_s]:
            forward -= 1.0
        if keys[pygame.K_a]:
            strafe -= 1.0
        if keys[pygame.K_d]:
            strafe += 1.0
        
        self.player.move(forward, strafe)
        
        # 旋转
        rotate = 0.0
        if keys[pygame.K_LEFT]:
            rotate -= 1.0
        if keys[pygame.K_RIGHT]:
            rotate += 1.0
        
        self.player.rotate(rotate)
        
        # 切换渲染模式
        if keys[pygame.K_1]:
            self.render_mode = RenderMode.WIREFRAME
        if keys[pygame.K_2]:
            self.render_mode = RenderMode.SOLID
        if keys[pygame.K_3]:
            self.render_mode = RenderMode.TEXTURED
        if keys[pygame.K_4]:
            self.render_mode = RenderMode.DOOM_STYLE
        
        # 更新FPS
        self.frame_count += 1
        current_time = pygame.time.get_ticks()
        if current_time - self.last_time >= 1000:
            self.fps = self.frame_count
            self.frame_count = 0
            self.last_time = current_time
    
    def render_minimap(self):
        """渲染小地图"""
        map_size = 200
        cell_size = 10
        offset_x = SCREEN_WIDTH - map_size - 20
        offset_y = 20
        
        # 绘制地图背景
        pygame.draw.rect(screen, (0, 0, 0, 200), 
                         (offset_x - 5, offset_y - 5, 
                          map_size + 10, map_size + 10))
        
        # 绘制墙壁
        for wall in self.walls:
            x = offset_x + int(wall.x) * cell_size
            y = offset_y + int(wall.y) * cell_size
            pygame.draw.rect(screen, (100, 100, 100), 
                            (x, y, cell_size, cell_size))
            pygame.draw.rect(screen, (150, 150, 150), 
                            (x, y, cell_size, cell_size), 1)
        
        # 绘制玩家
        player_x = offset_x + int(self.player.x * cell_size)
        player_y = offset_y + int(self.player.y * cell_size)
        pygame.draw.circle(screen, (255, 255, 0), 
                          (player_x, player_y), 5)
        
        # 绘制玩家方向
        dir_length = 15
        dir_x = player_x + math.cos(self.player.angle) * dir_length
        dir_y = player_y + math.sin(self.player.angle) * dir_length
        pygame.draw.line(screen, (255, 200, 0), 
                        (player_x, player_y), (int(dir_x), int(dir_y)), 2)
        
        # 绘制视野范围
        fov = self.renderer.fov
        for angle_offset in [-fov/2, fov/2]:
            ray_angle = self.player.angle + angle_offset
            ray_x = player_x + math.cos(ray_angle) * dir_length * 1.5
            ray_y = player_y + math.sin(ray_angle) * dir_length * 1.5
            pygame.draw.line(screen, (255, 200, 0, 128), 
                           (player_x, player_y), (int(ray_x), int(ray_y)), 1)
    
    def render_info(self):
        """渲染信息面板"""
        font = pygame.font.Font(None, 24)
        
        info = [
            f"位置: ({self.player.x:.2f}, {self.player.y:.2f})",
            f"方向: {math.degrees(self.player.angle):.1f}°",
            f"速度: {self.player.get_speed():.2f}",
            f"FPS: {self.fps}",
            f"渲染模式: {self.render_mode.name}",
            f"墙壁数量: {len(self.walls)}",
            "",
            "控制说明:",
            "W/A/S/D - 移动",
            "左右方向键 - 旋转",
            "1/2/3/4 - 切换渲染模式",
            "",
            "技术特性:",
            "• 多普勒效应: 距离感知颜色变化",
            "• 噪声算法: 纹理生成",
            "• DOOM渲染: 射线投射",
            "• 透视画法: 3D投影"
        ]
        
        for i, text in enumerate(info):
            text_surface = font.render(text, True, (255, 255, 200))
            screen.blit(text_surface, (20, 20 + i * 25))
    
    def render(self):
        """渲染游戏世界"""
        # 清屏
        screen.fill((0, 0, 0))
        
        # 根据渲染模式选择渲染方法
        if self.render_mode == RenderMode.DOOM_STYLE:
            self.renderer.render(self.player, self.walls)
            
            # 应用多普勒运动模糊
            if self.player.get_speed() > 0.1:
                temp_surface = screen.copy()
                screen.fill((0, 0, 0))
                DopplerEffect.apply_motion_blur(temp_surface, 
                                               self.player.get_speed(), 
                                               self.player.angle)
        
        # 渲染小地图和信息
        self.render_minimap()
        self.render_info()
        
        # 更新显示
        pygame.display.flip()

def main():

关键技术实现说明
1. 多普勒效应 (DopplerEffect类)
* 视觉化多普勒: 将距离信息转换为颜色变化
    * 近距离: 暖色（红/黄），模拟高频（蓝移）
    * 远距离: 冷色（蓝/紫），模拟低频（红移）
* 运动模糊: 根据玩家速度应用模糊效果，模拟高速运动时的感知
* 距离感知: 通过颜色渐变帮助玩家判断物体距离
2. 噪声算法 (NoiseAlgorithm类)
* Perlin噪声: 生成自然的纹理变化
* 分形噪声: 多重八度噪声，创建更丰富的纹理细节
* 动态纹理生成: 实时生成墙面纹理，无需预加载资源
* 表面噪点: 为渲染添加自然噪声，增强真实感
3. DOOM渲染原理 (DOOMRenderer类)
* 射线投射算法: 从玩家位置发射射线检测碰撞
* DDA算法: 数字微分分析，高效遍历网格
* 纹理映射: 将2D纹理映射到3D墙面
* 列渲染: 垂直列渲染，经典DOOM技术
4. 透视画法
* 透视投影: 正确模拟近大远小
* 视野限制: 90度视野范围
* 深度缓冲: 通过距离计算墙高
* 遮挡剔除: 只渲染可见表面

class AudioReceiver:
    """音频接收器 - 在玩家位置接收环境噪声"""
    
    def __init__(self, x: float, y: float, sample_rate: int = 44100, buffer_size: int = 512):
        self.x = x
        self.y = y
        self.sample_rate = sample_rate
        self.buffer_size = buffer_size
        self.audio_buffer = deque(maxlen=buffer_size)
        self.reverb_buffer = deque(maxlen=buffer_size * 4)
        self.noise_level = 0.1
        self.distance_factor = 1.0
        
        # 初始化音频流
        self.init_audio()
        
    def init_audio(self):
        """初始化音频处理"""
        # 填充初始噪声
        for _ in range(self.buffer_size):
            self.audio_buffer.append(random.uniform(-1, 1) * self.noise_level)
            
    def receive_noise(self, distance: float, velocity: float) -> float:
        """
        接收环境噪声
        distance: 到声源的距离
        velocity: 相对速度
        返回: 噪声样本
        """
        # 根据距离衰减噪声
        distance_att = 1.0 / (1.0 + distance * 0.5)
        
        # 多普勒频移效应
        doppler_shift = 1.0
        if abs(velocity) > 0.1:
            doppler_shift = 1.0 + (velocity * 0.1)
            
        # 生成噪声
        noise = random.uniform(-1, 1) * self.noise_level
        noise *= distance_att
        
        # 添加混响
        reverb_noise = 0
        if len(self.reverb_buffer) > 0:
            reverb_noise = self.reverb_buffer[0] * 0.3
            
        total_noise = noise + reverb_noise
        
        # 更新混响缓冲区
        self.reverb_buffer.append(total_noise)
        
        return total_noise * doppler_shift
    
    def update_position(self, x: float, y: float):
        """更新接收器位置"""
        self.x = x
        self.y = y

class KarplusStrongSynthesizer:
    """
    Karplus-Strong弦乐合成算法实现
    用接收到的噪声填充延迟线，循环读取并应用低通滤波
    """
    
    def __init__(self, sample_rate: int = 44100, buffer_size: int = 512):
        self.sample_rate = sample_rate
        self.delay_line_size = 100  # 延迟线大小
        self.delay_line = deque(maxlen=self.delay_line_size)
        self.read_ptr = 0
        self.write_ptr = 0
        self.feedback = 0.996  # 反馈系数
        self.lowpass_factor = 0.5  # 低通滤波系数
        self.output_buffer = deque(maxlen=buffer_size)
        
        # 初始化延迟线
        for _ in range(self.delay_line_size):
            self.delay_line.append(0.0)
            
    def fill_with_noise(self, noise_sample: float):
        """用噪声填充延迟线"""
        # 添加噪声到延迟线
        self.delay_line.append(noise_sample)
        
    def process(self) -> float:
        """
        处理一个样本
        返回: 合成的音频样本
        """
        if len(self.delay_line) < 2:
            return 0.0
            
        # 从延迟线读取
        sample1 = self.delay_line[0]
        
        # 应用低通滤波（平均滤波）
        filtered = sample1 * 0.5 + self.delay_line[1] * 0.5
        
        # 应用衰减
        filtered *= self.feedback
        
        # 添加回延迟线
        self.delay_line.append(filtered)
        
        # 移除旧的样本
        self.delay_line.popleft()
        
        return filtered
    
    def process_buffer(self, noise_samples: List[float]) -> List[float]:
        """处理一批噪声样本"""
        output_samples = []
        
        for noise in noise_samples:
            # 用噪声填充延迟线
            self.fill_with_noise(noise)
            
            # 处理并获取输出
            output = self.process()
            output_samples.append(output)
            
        return output_samples

class DopplerAudioEngine:
    """多普勒音频引擎 - 集成Karplus-Strong算法"""
    
    def __init__(self, sample_rate: int = 44100, buffer_size: int = 512):
        self.sample_rate = sample_rate
        self.buffer_size = buffer_size
        self.audio_receiver = None
        self.synthesizer = None
        self.audio_thread = None
        self.running = False
        self.audio_stream = None
        
        # 音频可视化数据
        self.audio_waveform = deque(maxlen=200)
        self.spectrum_data = deque(maxlen=100)
        
    def start(self, player_x: float, player_y: float):
        """启动音频引擎"""
        self.audio_receiver = AudioReceiver(player_x, player_y, 
                                           self.sample_rate, self.buffer_size)
        self.synthesizer = KarplusStrongSynthesizer(self.sample_rate, 
                                                   self.buffer_size)
        self.running = True
        
        # 启动音频线程
        self.audio_thread = threading.Thread(target=self.audio_loop)
        self.audio_thread.daemon = True
        self.audio_thread.start()
        
    def stop(self):
        """停止音频引擎"""
        self.running = False
        if self.audio_thread:
            self.audio_thread.join(timeout=1.0)
            
    def audio_loop(self):
        """音频处理主循环"""
        try:
            # 初始化音频流
            def audio_callback(outdata, frames, time_info, status):
                if status:
                    print(f"Audio status: {status}")
                    
                # 生成音频数据
                samples = []
                for _ in range(frames):
                    # 接收环境噪声（这里模拟距离和速度）
                    distance = random.uniform(1.0, 10.0)
                    velocity = random.uniform(-0.5, 0.5)
                    
                    # 获取噪声样本
                    noise_sample = self.audio_receiver.receive_noise(distance, velocity)
                    
                    # Karplus-Strong处理
                    output_sample = self.synthesizer.process()
                    
                    # 混合噪声和合成信号
                    mixed_sample = noise_sample * 0.3 + output_sample * 0.7
                    samples.append([mixed_sample, mixed_sample])
                    
                    # 记录波形数据用于可视化
                    self.audio_waveform.append(mixed_sample)
                    
                # 转换为numpy数组
                outdata[:] = np.array(samples, dtype=np.float32)
                
            # 打开音频流
            with sd.OutputStream(samplerate=self.sample_rate,
                               blocksize=self.buffer_size,
                               channels=2,
                               callback=audio_callback,
                               dtype='float32'):
                print("Audio engine started")
                
                while self.running:
                    time.sleep(0.1)
                    
        except Exception as e:
            print(f"Audio engine error: {e}")
            
    def update_receiver_position(self, x: float, y: float):
        """更新音频接收器位置"""
        if self.audio_receiver:
            self.audio_receiver.update_position(x, y)
            
    def visualize_audio(self, surface, x: int, y: int, width: int, height: int):
        """可视化音频波形和频谱"""
        if not self.audio_waveform:
            return
            
        # 绘制波形背景
        pygame.draw.rect(surface, (20, 20, 30), (x, y, width, height))
        pygame.draw.rect(surface, (40, 40, 60), (x, y, width, height), 1)
        
        # 绘制波形
        if len(self.audio_waveform) > 1:
            points = []
            for i, sample in enumerate(self.audio_waveform):
                if i >= width:
                    break
                px = x + i
                py = y + height // 2 - int(sample * (height // 2) * 0.8)
                points.append((px, py))
                
            if len(points) > 1:
                pygame.draw.lines(surface, (100, 200, 255), False, points, 2)
                
        # 绘制频谱
        spectrum_x = x
        spectrum_y = y + height + 10
        spectrum_height = 60
        
        pygame.draw.rect(surface, (20, 20, 30), 
                        (spectrum_x, spectrum_y, width, spectrum_height))
        
        # 简单的频谱模拟
        spectrum_bars = 20
        bar_width = width // spectrum_bars
        
        for i in range(spectrum_bars):
            bar_height = int((math.sin(i * 0.5 + time.time() * 5) * 0.5 + 0.5) * spectrum_height * 0.8)
            bar_x = spectrum_x + i * bar_width
            bar_y = spectrum_y + spectrum_height - bar_height
            
            # 根据频率设置颜色
            color_factor = i / spectrum_bars
            r = int(100 + color_factor * 155)
            g = int(100 + (1 - color_factor) * 155)
            b = 200
            
            pygame.draw.rect(surface, (r, g, b), 
                            (bar_x, bar_y, bar_width - 1, bar_height))
            
        # 绘制标题
        font = pygame.font.Font(None, 20)
        title = font.render("Karplus-Strong Audio Synthesis", True, (200, 200, 255))
        surface.blit(title, (x, y - 20))

class Vector2:
    """二维向量"""
    def __init__(self, x: float, y: float):
        self.x = x
        self.y = y
    
    def distance_to(self, other: 'Vector2') -> float:
        """计算到另一个点的距离"""
        return math.sqrt((self.x - other.x) ** 2 + (self.y - other.y) ** 2)
    
    def __str__(self):
        return f"({self.x:.2f}, {self.y:.2f})"

# 更新玩家类以包含音频接收器
class Player:
    """玩家类"""
    def __init__(self, x: float, y: float, angle: float = 0.0):
        self.x = x
        self.y = y
        self.angle = angle
        self.velocity_x = 0.0
        self.velocity_y = 0.0
        self.speed = 0.1
        self.rotate_speed = 0.05
        self.height = 0.5
        self.audio_engine = None
        
    def set_audio_engine(self, audio_engine: DopplerAudioEngine):
        """设置音频引擎"""
        self.audio_engine = audio_engine
        
    def move(self, forward: float, strafe: float):
        """移动玩家"""
        # 计算移动方向
        move_x = math.cos(self.angle) * forward + math.cos(self.angle + math.pi/2) * strafe
        move_y = math.sin(self.angle) * forward + math.sin(self.angle + math.pi/2) * strafe
        
        # 更新位置
        new_x = self.x + move_x * self.speed
        new_y = self.y + move_y * self.speed
        
        # 更新速度（用于多普勒效应）
        self.velocity_x = (new_x - self.x) * 60  # 乘以帧率近似速度
        self.velocity_y = (new_y - self.y) * 60
        
        self.x = new_x
        self.y = new_y
        
        # 更新音频接收器位置
        if self.audio_engine:
            self.audio_engine.update_receiver_position(self.x, self.y)
    
    def rotate(self, direction: float):
        """旋转玩家"""
        self.angle += direction * self.rotate_speed
        
        # 归一化角度
        if self.angle > 2 * math.pi:
            self.angle -= 2 * math.pi
        elif self.angle < 0:
            self.angle += 2 * math.pi
    
    def get_speed(self) -> float:
        """获取速度大小"""
        return math.sqrt(self.velocity_x ** 2 + self.velocity_y ** 2)
    
    def get_distance_to_wall(self, wall_x: float, wall_y: float) -> float:
        """计算到墙壁的距离"""
        return math.sqrt((self.x - wall_x) ** 2 + (self.y - wall_y) ** 2)

class GameWorld:
    """游戏世界"""
    def __init__(self):
        self.player = Player(5.5, 5.5, 0.0)
        self.walls = []
        self.generate_maze(10, 10)
        
        # 音频引擎
        self.audio_engine = None
        
        # 尝试初始化音频引擎
        try:
            self.audio_engine = DopplerAudioEngine(sample_rate=44100, buffer_size=512)
            self.audio_engine.start(self.player.x, self.player.y)
            self.player.set_audio_engine(self.audio_engine)
            self.audio_enabled = True
        except Exception as e:
            print(f"Audio engine failed to start: {e}")
            self.audio_enabled = False
        
        # 渲染设置
        self.screen_width = SCREEN_WIDTH
        self.screen_height = SCREEN_HEIGHT
        self.half_height = SCREEN_HEIGHT // 2
        self.fov = math.pi / 2
        
        # 性能统计
        self.frame_count = 0
        self.fps = 0
        self.last_time = pygame.time.get_ticks()
        
        # 字体
        self.font = pygame.font.Font(None, 24)
        
    def generate_maze(self, width: int, height: int):
        """生成迷宫"""
        # 创建外墙
        for x in range(width):
            self.walls.append((x, 0))
            self.walls.append((x, height - 1))
        
        for y in range(height):
            self.walls.append((0, y))
            self.walls.append((width - 1, y))
        
        # 创建内墙
        for x in range(2, width - 2, 2):
            for y in range(2, height - 2, 2):
                if random.random() > 0.3:
                    if random.random() > 0.5:
                        self.walls.append((x, y))
                        if random.random() > 0.5:
                            self.walls.append((x + 1, y))
                    else:
                        self.walls.append((x, y))
                        if random.random() > 0.5:
                            self.walls.append((x, y + 1))
    
    def cast_ray(self, start_x: float, start_y: float, angle: float) -> Tuple[float, Optional[Tuple[float, float]]]:
        """
        投射一条射线，返回击中墙壁的距离和坐标
        """
        ray_dir_x = math.cos(angle)
        ray_dir_y = math.sin(angle)
        
        map_x = int(start_x)
        map_y = int(start_y)
        
        delta_dist_x = abs(1 / ray_dir_x) if ray_dir_x != 0 else 1e30
        delta_dist_y = abs(1 / ray_dir_y) if ray_dir_y != 0 else 1e30
        
        step_x = 1 if ray_dir_x >= 0 else -1
        step_y = 1 if ray_dir_y >= 0 else -1
        
        if ray_dir_x < 0:
            side_dist_x = (start_x - map_x) * delta_dist_x
        else:
            side_dist_x = (map_x + 1.0 - start_x) * delta_dist_x
            
        if ray_dir_y < 0:
            side_dist_y = (start_y - map_y) * delta_dist_y
        else:
            side_dist_y = (map_y + 1.0 - start_y) * delta_dist_y
        
        while True:
            if side_dist_x < side_dist_y:
                side_dist_x += delta_dist_x
                map_x += step_x
                side = 0
            else:
                side_dist_y += delta_dist_y
                map_y += step_y
                side = 1
            
            # 检查是否击中墙壁
            if (map_x, map_y) in self.walls:
                if side == 0:
                    distance = (map_x - start_x + (1 - step_x) / 2) / ray_dir_x
                else:
                    distance = (map_y - start_y + (1 - step_y) / 2) / ray_dir_y
                
                wall_x = map_x
                wall_y = map_y
                return distance, (wall_x, wall_y)
            
            # 距离限制
            if side_dist_x > 20 and side_dist_y > 20:
                return 20.0, None
    
    def render_column(self, x: int, distance: float, wall_pos: Tuple[float, float]):
        """
        渲染一列墙
        """
        if distance <= 0:
            return
            
        # 计算墙的高度
        wall_height = int(self.screen_height / distance)
        
        # 计算墙的顶部和底部
        wall_top = self.half_height - wall_height // 2
        wall_bottom = wall_top + wall_height
        
        # 限制在屏幕范围内
        wall_top = max(0, wall_top)
        wall_bottom = min(self.screen_height, wall_bottom)
        
        if wall_top >= wall_bottom:
            return
        
        # 根据距离计算颜色
        if distance < 3:
            color = (200, 100, 100)  # 近处 - 红色
        elif distance < 7:
            color = (100, 200, 100)  # 中等 - 绿色
        else:
            color = (100, 100, 200)  # 远处 - 蓝色
            
        # 根据距离调整亮度
        brightness = max(0.3, 1.0 - distance / 20.0)
        r, g, b = color
        r = int(r * brightness)
        g = int(g * brightness)
        b = int(b * brightness)
        
        # 绘制墙
        pygame.draw.line(screen, (r, g, b), 
                        (x, wall_top), (x, wall_bottom), 2)
    
    def update(self):
        """更新游戏状态"""
        keys = pygame.key.get_pressed()
        
        # 移动
        forward = 0.0
        strafe = 0.0
        
        if keys[pygame.K_w]:
            forward += 1.0
        if keys[pygame.K_s]:
            forward -= 1.0
        if keys[pygame.K_a]:
            strafe -= 1.0
        if keys[pygame.K_d]:
            strafe += 1.0
        
        self.player.move(forward, strafe)
        
        # 旋转
        rotate = 0.0
        if keys[pygame.K_LEFT]:
            rotate -= 1.0
        if keys[pygame.K_RIGHT]:
            rotate += 1.0
        
        self.player.rotate(rotate)
        
        # 更新FPS
        self.frame_count += 1
        current_time = pygame.time.get_ticks()
        if current_time - self.last_time >= 1000:
            self.fps = self.frame_count
            self.frame_count = 0
            self.last_time = current_time
    
    def render(self):
        """渲染游戏世界"""
        # 清屏
        screen.fill((0, 0, 0))
        
        # 绘制天空和地面
        pygame.draw.rect(screen, (30, 30, 60), (0, 0, self.screen_width, self.half_height))
        pygame.draw.rect(screen, (20, 40, 20), (0, self.half_height, self.screen_width, self.half_height))
        
        # 射线投射渲染
        num_rays = self.screen_width // 2
        for i in range(num_rays):
            ray_angle = (i / num_rays - 0.5) * self.fov + self.player.angle
            distance, wall_pos = self.cast_ray(self.player.x, self.player.y, ray_angle)
            
            if wall_pos:
                x = i * 2
                self.render_column(x, distance, wall_pos)
                self.render_column(x + 1, distance, wall_pos)
                
                # 如果音频引擎可用，根据墙壁距离调整音频参数
                if self.audio_engine and wall_pos:
                    wall_x, wall_y = wall_pos
                    distance_to_wall = self.player.get_distance_to_wall(wall_x, wall_y)
                    # 可以在这里添加根据距离调整音频参数的逻辑
        
        # 渲染音频可视化
        if self.audio_enabled and self.audio_engine:
            self.audio_engine.visualize_audio(screen, 20, SCREEN_HEIGHT - 200, 300, 100)
        
        # 渲染小地图
        self.render_minimap()
        
        # 渲染信息

1. 初始化延迟线：用噪声（白噪声）填充
2. 循环处理：
    * 从延迟线读取样本
    * 应用低通滤波：y[n] = 0.5 * (x[n] + x[n-1])
    * 应用衰减：y[n] *= feedback (0.996)
    * 写回延迟线
3. 输出信号：衰减的准周期弦乐信号

def apply_doppler_effect(base_frequency, distance, relative_velocity):
    # 距离衰减
    attenuation = 1.0 / (1.0 + distance * 0.5)
    
    # 多普勒频移
    if relative_velocity > 0:  # 靠近
        frequency = base_frequency * (343.0 / (343.0 - relative_velocity))
    else:  # 远离
        frequency = base_frequency * (343.0 / (343.0 + abs(relative_velocity)))
    
    return frequency * attenuation

class DualRaySonar:
    """
    双射线声纳系统
    向同一方向发射两道时间间隔Δt的射线
    基于返回结果的绝对差值生成音频信号
    """
    
    def __init__(self, x: float, y: float, base_angle: float, 
                 time_interval: float = 0.001, sample_rate: int = 44100):
        """
        初始化声纳系统
        time_interval: 两道射线的时间间隔（秒）
        sample_rate: 音频采样率
        """
        self.x = x
        self.y = y
        self.base_angle = base_angle
        self.time_interval = time_interval
        self.sample_rate = sample_rate
        
        # 音频信号缓冲区
        self.audio_buffer = deque(maxlen=sample_rate)  # 1秒的缓冲区
        self.last_distance_1 = 0.0
        self.last_distance_2 = 0.0
        self.last_diff = 0.0
        
        # 信号处理参数
        self.smoothing_factor = 0.8
        self.noise_threshold = 0.01
        
    def emit_rays(self, angle: float, walls: List[Tuple[float, float]]) -> Tuple[float, float]:
        """
        发射两道射线，返回两个距离
        第一道射线：当前时间
        第二道射线：Δt时间后
        """
        # 第一道射线
        dist1 = self.cast_ray(angle, walls)
        
        # 模拟时间间隔后的第二道射线
        # 在实际系统中，这会真的等待Δt时间
        dist2 = self.cast_ray(angle, walls)
        
        # 添加一些变化来模拟运动
        # 如果玩家在移动，第二道射线的距离会略有不同
        movement_factor = random.uniform(0.99, 1.01)
        dist2 *= movement_factor
        
        return dist1, dist2
    
    def cast_ray(self, angle: float, walls: List[Tuple[float, float]]) -> float:
        """
        向指定方向发射射线，返回最短距离
        """
        ray_dir_x = math.cos(angle)
        ray_dir_y = math.sin(angle)
        
        min_distance = float('inf')
        
        for wall_x, wall_y in walls:
            # 简化的射线-点距离计算
            # 在实际系统中，这应该计算射线与墙壁的交点
            dx = wall_x - self.x
            dy = wall_y - self.y
            
            # 计算点到射线的距离
            # 点积公式: distance = |(P - A) × dir| / |dir|
            # 其中A是射线起点，dir是射线方向
            cross_product = abs(dx * ray_dir_y - dy * ray_dir_x)
            distance = cross_product  # 因为|dir| = 1
            
            if distance < min_distance and distance > 0.1:
                min_distance = distance
        
        # 如果没有击中任何墙壁，返回最大距离
        if min_distance == float('inf'):
            min_distance = 20.0
            
        return min_distance
    
    def generate_audio_sample(self, dist1: float, dist2: float) -> float:
        """
        基于两个距离的绝对差值生成音频样本
        """
        # 计算绝对差值
        diff = abs(dist1 - dist2)
        
        # 应用噪声阈值
        if diff < self.noise_threshold:
            diff = 0.0
        
        # 平滑处理
        smoothed_diff = (self.last_diff * self.smoothing_factor + 
                        diff * (1 - self.smoothing_factor))
        
        # 归一化到[-1, 1]范围
        # 假设最大差值为5.0
        max_diff = 5.0
        normalized = (smoothed_diff / max_diff) * 2.0 - 1.0
        
        # 应用非线性变换使信号更有音乐性
        if normalized > 0:
            audio_sample = math.sqrt(normalized)
        else:
            audio_sample = -math.sqrt(abs(normalized))
        
        # 保存状态
        self.last_distance_1 = dist1
        self.last_distance_2 = dist2
        self.last_diff = smoothed_diff
        
        return audio_sample
    
    def update(self, x: float, y: float, angle: float, walls: List[Tuple[float, float]]):
        """
        更新声纳位置并发射射线
        """
        self.x = x
        self.y = y
        self.base_angle = angle
        
        # 发射射线
        dist1, dist2 = self.emit_rays(angle, walls)
        
        # 生成音频样本
        audio_sample = self.generate_audio_sample(dist1, dist2)
        
        # 添加到缓冲区
        self.audio_buffer.append(audio_sample)
        
        return audio_sample, dist1, dist2
    
    def get_audio_buffer(self, num_samples: int) -> List[float]:
        """
        获取指定数量的音频样本
        """
        if len(self.audio_buffer) < num_samples:
            # 如果缓冲区不足，用零填充
            zeros_needed = num_samples - len(self.audio_buffer)
            return list(self.audio_buffer) + [0.0] * zeros_needed
        else:
            return list(self.audio_buffer)[-num_samples:]

class EnhancedKarplusStrong:
    """
    增强版Karplus-Strong合成器
    用声纳信号填充延迟线，生成准周期音频信号
    """
    
    def __init__(self, sample_rate: int = 44100, buffer_size: int = 512):
        self.sample_rate = sample_rate
        self.delay_line_size = sample_rate // 220  # 对应220Hz的频率
        self.delay_line = deque(maxlen=self.delay_line_size)
        self.read_ptr = 0
        self.feedback = 0.998  # 非常接近1，使声音持续更久
        self.lowpass_factor = 0.5
        self.output_buffer = deque(maxlen=buffer_size)
        
        # 初始化延迟线为0
        for _ in range(self.delay_line_size):
            self.delay_line.append(0.0)
        
        # 频率调制参数
        self.frequency_mod = 1.0
        self.mod_depth = 0.1
        self.mod_rate = 0.5
        
    def fill_with_sonar_signal(self, sonar_samples: List[float]):
        """
        用声纳信号填充延迟线
        每次用新的声纳样本替换延迟线的一部分
        """
        if not sonar_samples:
            return
            
        # 将声纳样本写入延迟线的不同位置
        for i, sample in enumerate(sonar_samples):
            if i < self.delay_line_size:
                # 替换延迟线中的样本
                self.delay_line[i] = sample
            else:
                break
                
    def process(self, modulation: float = 0.0) -> float:
        """
        处理一个样本，应用低通滤波和衰减
        modulation: 频率调制信号，范围[0, 1]
        """
        if len(self.delay_line) < 2:
            return 0.0
        
        # 应用频率调制
        effective_feedback = self.feedback * (1.0 + modulation * self.mod_depth)
        
        # 从延迟线读取
        sample1 = self.delay_line[self.read_ptr]
        
        # 应用低通滤波（平均滤波）
        next_idx = (self.read_ptr + 1) % len(self.delay_line)
        sample2 = self.delay_line[next_idx]
        filtered = sample1 * 0.5 + sample2 * 0.5
        
        # 应用衰减
        filtered *= effective_feedback
        
        # 写回延迟线
        self.delay_line[self.read_ptr] = filtered
        
        # 移动读指针
        self.read_ptr = (self.read_ptr + 1) % len(self.delay_line)
        
        return filtered
    
    def process_buffer(self, sonar_samples: List[float], 
                      modulation_buffer: List[float] = None) -> List[float]:
        """
        处理一批样本
        """
        if not modulation_buffer:
            modulation_buffer = [0.0] * len(sonar_samples)
            
        output_samples = []
        
        # 用声纳信号填充延迟线
        self.fill_with_sonar_signal(sonar_samples)
        
        # 处理每个样本
        for i, modulation in enumerate(modulation_buffer):
            output = self.process(modulation)
            output_samples.append(output)
            
        return output_samples

class SonarAudioEngine:
    """
    完整的音频系统：双射线声纳 + Karplus-Strong合成
    """
    
    def __init__(self, sample_rate: int = 44100, buffer_size: int = 512):
        self.sample_rate = sample_rate
        self.buffer_size = buffer_size
        self.sonar = None
        self.synthesizer = None
        self.audio_thread = None
        self.running = False
        
        # 音频流
        self.audio_stream = None
        
        # 可视化数据
        self.sonar_waveform = deque(maxlen=200)
        self.ks_waveform = deque(maxlen=200)
        self.distance_history = deque(maxlen=100)
        self.diff_history = deque(maxlen=100)
        
        # 统计信息
        self.audio_stats = {
            'samples_processed': 0,
            'avg_frequency': 220.0,
            'current_amplitude': 0.0
        }
        
    def start(self, player_x: float, player_y: float, 
              player_angle: float, walls: List[Tuple[float, float]]):
        """启动音频引擎"""
        self.sonar = DualRaySonar(player_x, player_y, player_angle, 
                                 time_interval=0.001, 
                                 sample_rate=self.sample_rate)
        self.synthesizer = EnhancedKarplusStrong(self.sample_rate, 
                                                self.buffer_size)
        self.running = True
        
        # 预热延迟线
        initial_noise = [random.uniform(-1, 1) * 0.1 for _ in range(100)]
        self.synthesizer.fill_with_sonar_signal(initial_noise)
        
        # 启动音频线程
        self.audio_thread = threading.Thread(target=self.audio_loop, 
                                           args=(walls,))
        self.audio_thread.daemon = True
        self.audio_thread.start()
        
    def stop(self):
        """停止音频引擎"""
        self.running = False
        if self.audio_thread:
            self.audio_thread.join(timeout=1.0)
            
    def audio_loop(self, walls: List[Tuple[float, float]]):
        """音频处理主循环"""
        try:
            def audio_callback(outdata, frames, time_info, status):
                if status:
                    print(f"Audio status: {status}")
                
                # 生成音频数据
                samples = []
                for i in range(frames):
                    # 更新声纳（模拟实时更新）
                    if self.sonar and i % 10 == 0:  # 每10个样本更新一次声纳
                        # 模拟轻微的角度变化
                        mod_angle = self.sonar.base_angle + math.sin(time.time() * 0.5) * 0.1
                        sonar_sample, dist1, dist2 = self.sonar.update(
                            self.sonar.x, self.sonar.y, mod_angle, walls
                        )
                        
                        # 记录数据用于可视化
                        self.distance_history.append((dist1, dist2))
                        self.diff_history.append(abs(dist1 - dist2))
                        self.sonar_waveform.append(sonar_sample)
                    
                    # Karplus-Strong处理
                    if self.synthesizer:
                        # 生成调制信号
                        mod_signal = math.sin(time.time() * self.synthesizer.mod_rate) * 0.3
                        
                        # 处理样本
                        ks_sample = self.synthesizer.process(mod_signal)
                        self.ks_waveform.append(ks_sample)
                        
                        # 混合声纳原始信号和KS合成信号
                        mixed_sample = ks_sample
                        
                        # 应用音量控制
                        mixed_sample *= 0.5
                        
                        samples.append([mixed_sample, mixed_sample])
                        
                        # 更新统计
                        self.audio_stats['samples_processed'] += 1
                        self.audio_stats['current_amplitude'] = abs(mixed_sample)
                    else:
                        samples.append([0.0, 0.0])
                
                # 转换为numpy数组
                outdata[:] = np.array(samples, dtype=np.float32)
            
            # 打开音频流
            with sd.OutputStream(samplerate=self.sample_rate,
                               blocksize=self.buffer_size,
                               channels=2,
                               callback=audio_callback,
                               dtype='float32'):
                print("Sonar audio engine started")
                
                while self.running:
                    time.sleep(0.1)
                    
        except Exception as e:
            print(f"Audio engine error: {e}")

    def generate_maze(self, width: int, height: int):
        """生成迷宫"""
        # 创建外墙
        for x in range(width):
            self.walls.append((x, 0))
            self.walls.append((x, height - 1))
        
        for y in range(height):
            self.walls.append((0, y))
            self.walls.append((width - 1, y))
        
        # 创建内墙
        for x in range(2, width - 2, 2):
            for y in range(2, height - 2, 2):
                if random.random() > 0.3:
                    if random.random() > 0.5:
                        self.walls.append((x, y))
                        if random.random() > 0.5:
                            self.walls.append((x + 1, y))
                    else:
                        self.walls.append((x, y))
                        if random.random() > 0.5:
                            self.walls.append((x, y + 1))
    
    def cast_ray(self, start_x: float, start_y: float, angle: float) -> float:
        """投射一条射线，返回最短距离"""
        ray_dir_x = math.cos(angle)
        ray_dir_y = math.sin(angle)
        
        min_distance = 20.0
        
        for wall_x, wall_y in self.walls:
            # 简化的距离计算
            dx = wall_x - start_x
            dy = wall_y - start_y
            distance = math.sqrt(dx*dx + dy*dy)
            
            # 检查是否在射线方向上
            dot_product = dx * ray_dir_x + dy * ray_dir_y
            if dot_product > 0 and distance < min_distance:
                min_distance = distance
                
        return min_distance

系统架构详细说明
1. 双射线声纳系统 (DualRaySonar)
class DualRaySonar:
    def emit_rays(self, angle, walls):
        # 发射第一道射线
        dist1 = self.cast_ray(angle, walls)
        
        # 模拟时间间隔Δt后的第二道射线
        dist2 = self.cast_ray(angle, walls) * movement_factor
        
        return dist1, dist2
    
    def generate_audio_sample(self, dist1, dist2):
        # 计算绝对差值
        diff = abs(dist1 - dist2)
        
        # 应用噪声阈值和平滑
        smoothed_diff = (self.last_diff * smoothing + diff * (1 - smoothing))
        
        # 转换为音频信号
        audio_signal = convert_to_audio(smoothed_diff)
        
        return audio_signal

工作原理：
* 向同一方向发射两道时间间隔Δt的射线
* 测量每道射线到最近墙壁的距离
* 计算距离的绝对差值：Δd = |d₁ - d₂|
* 将Δd转换为音频信号（归一化到[-1, 1]范围）
2. Karplus-Strong增强合成器
class EnhancedKarplusStrong:
    def fill_with_sonar_signal(self, sonar_samples):
        # 用声纳信号填充延迟线
        for i, sample in enumerate(sonar_samples):
            if i < self.delay_line_size:
                self.delay_line[i] = sample
    
    def process(self, modulation):
        # 从延迟线读取样本
        sample1 = self.delay_line[self.read_ptr]
        
        # 低通滤波：平均当前样本和下一个样本
        filtered = sample1 * 0.5 + sample2 * 0.5
        
        # 应用衰减反馈
        filtered *= self.feedback * (1 + modulation)
        
        # 写回延迟线
        self.delay_line[self.read_ptr] = filtered
        
        # 移动指针
        self.read_ptr = (self.read_ptr + 1) % len(self.delay_line)
        
        return filtered

核心算法：
1. 延迟线初始化：用声纳信号填充
2. 循环读取：从延迟线中循环读取样本
3. 低通滤波：y[n] = 0.5 * (x[n] + x[n+1])
4. 衰减反馈：y[n] *= feedback
5. 准周期输出：产生类似弦乐的衰减信号
3. 完整音频系统工作流程
双射线声纳 → 距离差计算 → 音频转换 → Karplus-Strong处理 → 音频输出
     ↓           ↓           ↓             ↓              ↓
  发射射线     Δd = |d1-d2|  信号归一化    延迟线填充     最终音频
               ↓           ↓           ↓             ↓
             距离感知     电信号      低通滤波     准周期信号

数学原理
1. 声纳信号生成
Δd = |d₁ - d₂|                    # 距离绝对差值
V = (Δd / Δd_max) * 2 - 1         # 归一化到[-1, 1]
V_smooth = α·V_prev + (1-α)·V    # 指数平滑

2. Karplus-Strong算法
初始化: delay_line[0:N] = sonar_signal
循环:
  y[n] = delay_line[p]
  y_filtered = 0.5 * (y[n] + delay_line[(p+1)%N])
  delay_line[p] = feedback * y_filtered
  p = (p + 1) % N
  输出: y[n]

class PhysicsRaySonar:
    """
    物理精确的双射线声纳系统
    基于真实物理：运动产生声音，静止无声
    """
    
    def __init__(self, x: float, y: float, base_angle: float, 
                 sample_rate: int = 44100):
        """
        初始化物理声纳
        """
        self.x = x
        self.y = y
        self.base_angle = base_angle
        self.sample_rate = sample_rate
        
        # 音频缓冲区
        self.audio_buffer = deque(maxlen=1000)
        self.raw_distance_diff = deque(maxlen=100)
        self.environment_density = deque(maxlen=100)
        
        # 物理参数
        self.last_position = (x, y)
        self.last_time = time.time()
        self.velocity = 0.0
        self.acceleration = 0.0
        
        # 声学参数
        self.speed_of_sound = 343.0  # m/s
        self.min_velocity_threshold = 0.05  # 最小运动速度阈值
        self.max_velocity = 5.0
        
        # 信号处理
        self.smoothing_alpha = 0.9
        self.last_audio_sample = 0.0
        
    def calculate_motion(self, new_x: float, new_y: float) -> Tuple[float, float]:
        """
        计算玩家的运动参数
        返回：(速度, 加速度)
        """
        current_time = time.time()
        dt = current_time - self.last_time
        
        if dt <= 0:
            return 0.0, 0.0
        
        # 计算位移
        dx = new_x - self.last_position[0]
        dy = new_y - self.last_position[1]
        displacement = math.sqrt(dx*dx + dy*dy)
        
        # 计算速度
        new_velocity = displacement / dt
        
        # 计算加速度
        new_acceleration = (new_velocity - self.velocity) / dt
        
        # 更新状态
        self.last_position = (new_x, new_y)
        self.last_time = current_time
        self.velocity = new_velocity
        self.acceleration = new_acceleration
        
        return new_velocity, new_acceleration
    
    def cast_ray_with_motion(self, angle: float, velocity: float, 
                            walls: List[Tuple[float, float]], 
                            is_second_ray: bool = False) -> float:
        """
        考虑运动发射射线
        is_second_ray: True表示第二道射线（考虑了运动位移）
        """
        # 第一道射线：当前位置
        # 第二道射线：考虑速度位移后的位置
        if is_second_ray and velocity > self.min_velocity_threshold:
            # 模拟时间间隔Δt后的位置
            dt = 0.001  # 1ms时间间隔
            displacement = velocity * dt
            
            # 计算位移向量
            dx = math.cos(angle) * displacement
            dy = math.sin(angle) * displacement
            
            # 第二道射线的发射位置
            ray_x = self.x + dx
            ray_y = self.y + dy
        else:
            # 第一道射线或静止时的射线
            ray_x = self.x
            ray_y = self.y
        
        # 发射射线，返回最近墙壁距离
        return self._cast_single_ray(ray_x, ray_y, angle, walls)
    
    def _cast_single_ray(self, start_x: float, start_y: float, 
                        angle: float, walls: List[Tuple[float, float]]) -> float:
        """
        发射单道射线，返回最近墙壁距离
        """
        ray_dir_x = math.cos(angle)
        ray_dir_y = math.sin(angle)
        
        min_distance = 20.0  # 最大距离
        
        for wall_x, wall_y in walls:
            # 简化的射线-墙壁距离计算
            dx = wall_x - start_x
            dy = wall_y - start_y
            
            # 计算点到射线的垂直距离
            cross_product = abs(dx * ray_dir_y - dy * ray_dir_x)
            distance = cross_product
            
            # 确保点在射线正前方
            dot_product = dx * ray_dir_x + dy * ray_dir_y
            if dot_product > 0 and distance < min_distance:
                min_distance = distance
        
        return min_distance
    
    def calculate_environment_density(self, distances: List[float]) -> float:
        """
        计算环境密度
        空旷区域密度低，拥挤区域密度高
        """
        if not distances:
            return 0.0
        
        # 计算平均距离
        avg_distance = sum(distances) / len(distances)
        
        # 将距离转换为密度（距离越近，密度越高）
        # 使用反比例函数：密度 = 1 / (1 + 距离)
        density = 1.0 / (1.0 + avg_distance)
        
        return density
    
    def generate_motion_triggered_audio(self, velocity: float, 
                                       dist1: float, dist2: float,
                                       environment_density: float) -> float:
        """
        生成运动触发的音频信号
        静止时无声，运动时有声，空旷处声弱
        """
        # 1. 检查是否运动
        if velocity < self.min_velocity_threshold:
            # 静止，无声
            return 0.0
        
        # 2. 计算距离差（运动产生变化）
        distance_diff = abs(dist1 - dist2)
        
        # 3. 根据速度调整信号强度
        velocity_factor = min(1.0, velocity / self.max_velocity)
        
        # 4. 根据环境密度调整信号
        # 空旷区域（低密度）信号弱，拥挤区域（高密度）信号强
        density_factor = environment_density
        
        # 5. 生成基础音频信号
        if distance_diff > 0.001:  # 微小阈值避免噪声
            # 距离差产生信号
            base_signal = distance_diff * velocity_factor * density_factor
            
            # 添加一些物理特性
            # 运动越快，频率越高
            frequency_factor = 1.0 + velocity_factor * 2.0
            
            # 生成振荡信号
            t = time.time()
            oscillation = math.sin(t * frequency_factor * 10.0)
            
            audio_signal = base_signal * oscillation
        else:
            audio_signal = 0.0
        
        # 6. 平滑处理
        smoothed = (self.last_audio_sample * self.smoothing_alpha + 
                   audio_signal * (1 - self.smoothing_alpha))
        
        # 7. 限制幅度
        smoothed = max(-1.0, min(1.0, smoothed))
        
        self.last_audio_sample = smoothed
        return smoothed
    
    def update(self, x: float, y: float, angle: float, 
              walls: List[Tuple[float, float]]) -> Tuple[float, float, float]:
        """
        更新声纳并生成音频信号
        返回：(音频信号, 速度, 环境密度)
        """
        # 更新位置
        self.x = x
        self.y = y
        self.base_angle = angle
        
        # 计算运动参数
        velocity, acceleration = self.calculate_motion(x, y)
        
        # 向多个方向发射射线，计算环境密度
        num_direction_samples = 8
        direction_angles = [angle + i * (2 * math.pi / num_direction_samples) 
                          for i in range(num_direction_samples)]
        
        distances = []
        for dir_angle in direction_angles:
            dist = self._cast_single_ray(x, y, dir_angle, walls)
            distances.append(dist)
        
        # 计算环境密度
        density = self.calculate_environment_density(distances)
        
        # 只在有显著运动时发射双射线
        if velocity > self.min_velocity_threshold:
            # 发射双射线
            dist1 = self.cast_ray_with_motion(angle, velocity, walls, False)
            dist2 = self.cast_ray_with_motion(angle, velocity, walls, True)
            
            # 生成音频信号
            audio_signal = self.generate_motion_triggered_audio(
                velocity, dist1, dist2, density
            )
        else:
            # 静止，无声
            audio_signal = 0.0
            dist1 = dist2 = distances[0] if distances else 20.0
        
        # 记录数据
        if velocity > self.min_velocity_threshold:
            self.raw_distance_diff.append(abs(dist1 - dist2))
        self.environment_density.append(density)
        self.audio_buffer.append(audio_signal)
        
        return audio_signal, velocity, density

class AdaptiveKarplusStrong:
    """
    自适应Karplus-Strong合成器
    根据运动和环境自动调整参数
    """
    
    def __init__(self, sample_rate: int = 44100):
        self.sample_rate = sample_rate
        
        # 自适应延迟线
        self.base_delay_size = sample_rate // 220  # 220Hz基频
        self.delay_line = deque(maxlen=self.base_delay_size * 2)
        
        # 自适应参数
        self.feedback = 0.995
        self.lowpass_factor = 0.5
        
        # 状态
        self.read_ptr = 0
        self.is_active = False
        self.activation_threshold = 0.01
        
        # 初始化延迟线
        for _ in range(self.base_delay_size):
            self.delay_line.append(0.0)
    
    def adapt_parameters(self, velocity: float, density: float, audio_signal: float):
        """
        根据运动和环境自适应调整参数
        """
        # 检查是否需要激活
        self.is_active = abs(audio_signal) > self.activation_threshold
        
        if not self.is_active:
            return
        
        # 根据速度调整反馈（运动越快，衰减越慢）
        # 速度0-1映射到feedback 0.99-0.999
        velocity_factor = min(1.0, velocity / 5.0)
        self.feedback = 0.99 + velocity_factor * 0.009
        
        # 根据环境密度调整低通滤波
        # 空旷区域（低密度）高频更多，拥挤区域（高密度）低频更多
        self.lowpass_factor = 0.3 + density * 0.4  # 0.3-0.7
        
        # 根据运动强度调整延迟线大小（改变音高）
        intensity = min(1.0, velocity * density * 2.0)
        delay_scale = 0.8 + intensity * 0.4  # 0.8-1.2
        new_size = int(self.base_delay_size * delay_scale)
        
        # 调整延迟线大小
        if new_size != len(self.delay_line):
            while len(self.delay_line) < new_size:
                self.delay_line.append(0.0)
            while len(self.delay_line) > new_size:
                self.delay_line.pop()
    
    def fill_with_adaptive_signal(self, audio_signal: float, density: float):
        """
        用自适应信号填充延迟线
        """
        if not self.is_active:
            return
        
        # 根据环境密度调整信号强度
        # 空旷区域信号衰减更快
        adjusted_signal = audio_signal * (0.5 + density * 0.5)
        
        # 用调整后的信号替换延迟线中的随机位置
        if len(self.delay_line) > 0:
            # 在多个随机位置插入信号
            num_insertions = max(1, int(density * 5))
            for _ in range(num_insertions):
                idx = random.randint(0, len(self.delay_line) - 1)
                # 混合新旧信号
                self.delay_line[idx] = (self.delay_line[idx] * 0.3 + 
                                       adjusted_signal * 0.7)
    
    def process(self) -> float:
        """
        处理一个样本
        """
        if not self.is_active or len(self.delay_line) < 2:
            return 0.0
        
        # 从延迟线读取
        sample1 = self.delay_line[self.read_ptr]
        
        # 应用自适应低通滤波
        next_idx = (self.read_ptr + 1) % len(self.delay_line)
        sample2 = self.delay_line[next_idx]
        filtered = (sample1 * self.lowpass_factor + 
                   sample2 * (1 - self.lowpass_factor))
        
        # 应用衰减
        filtered *= self.feedback
        
        # 写回延迟线
        self.delay_line[self.read_ptr] = filtered
        
        # 移动读指针
        self.read_ptr = (self.read_ptr + 1) % len(self.delay_line)
        
        return filtered
    
    def get_synthesis_info(self) -> dict:
        """
        获取合成器状态信息
        """
        if len(self.delay_line) == 0:
            frequency = 0.0
        else:
            frequency = self.sample_rate / len(self.delay_line)
        
        return {
            'active': self.is_active,
            'frequency': frequency,
            'feedback': self.feedback,
            'lowpass': self.lowpass_factor,
            'delay_size': len(self.delay_line)
        }

class DynamicAudioSystem:
    """
    动态音频系统
    集成物理声纳和自适应合成器
    """
    
    def __init__(self, sample_rate: int = 44100, buffer_size: int = 512):
        self.sample_rate = sample_rate
        self.buffer_size = buffer_size
        
        # 系统组件
        self.sonar = None
        self.synthesizer = None
        self.audio_thread = None
        self.running = False
        
        # 音频流
        self.audio_stream = None
        
        # 可视化数据
        self.visualization_data = {
            'motion_signal': deque(maxlen=200),
            'synthesis_signal': deque(maxlen=200),
            'velocity': deque(maxlen=100),
            'density': deque(maxlen=100),
            'feedback': deque(maxlen=100),
            'frequency': deque(maxlen=100)
        }
        
        # 统计
        self.stats = {
            'total_samples': 0,
            'silent_frames': 0,
            'active_frames': 0,
            'max_velocity': 0.0,
            'avg_density': 0.0
        }
        
    def start(self, x: float, y: float, angle: float, walls: List[Tuple[float, float]]):
        """启动音频系统"""
        self.sonar = PhysicsRaySonar(x, y, angle, self.sample_rate)
        self.synthesizer = AdaptiveKarplusStrong(self.sample_rate)
        self.running = True
        
        # 启动音频线程
        self.audio_thread = threading.Thread(target=self._audio_processing_loop,
                                           args=(walls,))
        self.audio_thread.daemon = True
        self.audio_thread.start()
        
    def stop(self):
        """停止音频系统"""
        self.running = False
        if self.audio_thread:
            self.audio_thread.join(timeout=1.0)
    
    def update_position(self, x: float, y: float, angle: float):
        """更新玩家位置"""
        if self.sonar:
            self.sonar.x = x
            self.sonar.y = y
            self.sonar.base_angle = angle
    
    def _audio_processing_loop(self, walls: List[Tuple[float, float]]):
        """音频处理主循环"""
        try:
            def audio_callback(outdata, frames, time_info, status):
                if status:
                    print(f"Audio status: {status}")
                
                samples = []
                for i in range(frames):
                    # 更新声纳（每帧一次）
                    if i % 44 == 0:  # 每44个样本更新一次（~1ms @ 44.1kHz）
                        audio_signal, velocity, density = self.sonar.update(
                            self.sonar.x, self.sonar.y, self.sonar.base_angle, walls
                        )
                        
                        # 自适应调整合成器
                        self.synthesizer.adapt_parameters(velocity, density, audio_signal)
                        
                        # 填充延迟线
                        self.synthesizer.fill_with_adaptive_signal(audio_signal, density)
                        
                        # 记录数据
                        self.visualization_data['motion_signal'].append(audio_signal)
                        self.visualization_data['velocity'].append(velocity)
                        self.visualization_data['density'].append(density)
                        
                        # 更新统计
                        self.stats['total_samples'] += 1
                        if abs(audio_signal) < 0.001:
                            self.stats['silent_frames'] += 1
                        else:
                            self.stats['active_frames'] += 1
                        
                        self.stats['max_velocity'] = max(self.stats['max_velocity'], velocity)
                        
                        # 更新平均密度
                        if len(self.visualization_data['density']) > 0:
                            self.stats['avg_density'] = (
                                sum(self.visualization_data['density']) / 
                                len(self.visualization_data['density'])
                            )
                    
                    # 合成音频
                    ks_sample = self.synthesizer.process()
                    self.visualization_data['synthesis_signal'].append(ks_sample)
                    
                    # 获取合成器信息
                    synth_info = self.synthesizer.get_synthesis_info()
                    self.visualization_data['feedback'].append(synth_info['feedback'])
                    self.visualization_data['frequency'].append(synth_info['frequency'])
                    
                    # 应用音量控制
                    final_sample = ks_sample * 0.7
                    samples.append([final_sample, final_sample])
                
                outdata[:] = np.array(samples, dtype=np.float32)
            
            # 打开音频流
            with sd.OutputStream(samplerate=self.sample_rate,
                               blocksize=self.buffer_size,
                               channels=2,
                               callback=audio_callback,
                               dtype='float32'):
                print("Dynamic audio system started")
                
                while self.running:
                    time.sleep(0.1)
                    
        except Exception as e:
            print(f"Audio system error: {e}")

class GameWorld:
    def __init__(self):
        # 玩家状态
        self.player_x = 5.5
        self.player_y = 5.5
        self.player_angle = 0.0
        self.player_speed = 0.1
        self.turn_speed = 0.05
        
        # 环境
        self.walls = []
        self._generate_environment()
        
        # 音频系统
        self.audio_system = None
        self._init_audio_system()
        
        # 渲染
        self.screen_width = SCREEN_WIDTH
        self.screen_height = SCREEN_HEIGHT
        self.half_height = HEIGHT // 2
        self.fov = math.pi / 2
        
        # 性能
        self.frame_count = 0
        self.fps = 0
        self.last_time = pygame.time.get_ticks()
        
        # 字体
        self.font = pygame.font.Font(None, 24)
        
    def _generate_environment(self):
        """生成测试环境"""
        # 外框
        for x in range(10):
            self.walls.append((x, 0))
            self.walls.append((x, 9))
        
        for y in range(10):
            self.walls.append((0, y))
            self.walls.append((9, y))
        
        # 内部结构 - 创建不同密度的区域
        # 空旷区域 (中间)
        # 拥挤区域 (四周添加更多墙)
        for x in range(2, 8):
            for y in range(2, 8):
                if random.random() < 0.7:  # 70%概率添加墙
                    self.walls.append((x, y))
    
    def _init_audio_system(self):
        """初始化音频系统"""
        try:
            self.audio_system = DynamicAudioSystem(sample_rate=44100, buffer_size=512)
            self.audio_system.start(self.player_x, self.player_y, 
                                   self.player_angle, self.walls)
            self.audio_enabled = True
            print("动态音频系统启动成功")
        except Exception as e:
            print(f"音频系统启动失败: {e}")
            self.audio_enabled = False
    
    def update(self):
        """更新游戏状态"""
        keys = pygame.key.get_pressed()
        
        # 保存旧位置用于计算
        old_x, old_y = self.player_x, self.player_y
        
        # 移动
        if keys[pygame.K_w]:
            self.player_x += math.cos(self.player_angle) * self.player_speed
            self.player_y += math.sin(self.player_angle) * self.player_speed
        if keys[pygame.K_s]:
            self.player_x -= math.cos(self.player_angle) * self.player_speed
            self.player_y -= math.sin(self.player_angle) * self.player_speed
        
        # 转向
        if keys[pygame.K_a]:
            self.player_angle -= self.turn_speed
        if keys[pygame.K_d]:
            self.player_angle += self.turn_speed
        
        # 更新音频系统位置
        if self.audio_enabled and self.audio_system:
            self.audio_system.update_position(self.player_x, self.player_y, 
                                             self.player_angle)
        
        # 限制角度
        if self.player_angle > 2 * math.pi:
            self.player_angle -= 2 * math.pi
        elif self.player_angle < 0:
            self.player_angle += 2 * math.pi
        
        # 更新FPS
        self.frame_count += 1
        current_time = pygame.time.get_ticks()
        if current_time - self.last_time >= 1000:
            self.fps = self.frame_count
            self.frame_count = 0
            self.last_time = current_time
    
    def _cast_ray(self, x, y, angle):
        """简单的射线投射"""
        ray_dir_x = math.cos(angle)
        ray_dir_y = math.sin(angle)
        
        min_dist = 20.0
        
        for wx, wy in self.walls:
            dx = wx - x
            dy = wy - y
            dist = math.sqrt(dx*dx + dy*dy)
            
            # 检查方向
            dot = dx * ray_dir_x + dy * ray_dir_y
            if dot > 0 and dist < min_dist:
                min_dist = dist
        
        return min_dist
    
    def _render_column(self, x, distance):
        """渲染墙列"""
        if distance <= 0:
            return
        
        wall_height = int(self.screen_height / distance)
        wall_top = self.half_height - wall_height // 2
        wall_bottom = wall_top + wall_height
        
        wall_top = max(0, wall_top)
        wall_bottom = min(self.screen_height, wall_bottom)
        
        if wall_top >= wall_bottom:
            return
        
        # 根据距离着色
        if distance < 3:
            color = (200, 100, 100)
        elif distance < 6:
            color = (100, 200, 100)
        else:
            color = (100, 100, 200)
        
        brightness = max(0.3, 1.0 - distance / 20.0)
        r, g, b = [int(c * brightness) for c in color]
        
        pygame.draw.line(screen, (r, g, b), 
                        (x, wall_top), (x, wall_bottom), 2)
    
    def render(self):
        """渲染游戏"""
        # 清屏
        screen.fill((0, 0, 0))
        
        # 天空和地面
        pygame.draw.rect(screen, (40, 40, 80), (0, 0, SCREEN_WIDTH, self.half_height))
        pygame.draw.rect(screen, (30, 50, 30), (0, self.half_height, SCREEN_WIDTH, self.half_height))
        
        # 射线投射
        num_rays = SCREEN_WIDTH // 2
        for i in range(num_rays):
            ray_angle = (i / num_rays - 0.5) * self.fov + self.player_angle
            distance = self._cast_ray(self.player_x, self.player_y, ray_angle)
            
            col_x = i * 2
            self._render_column(col_x, distance)
            self._render_column(col_x + 1, distance)
        
        # 音频系统可视化
        if self.audio_enabled and self.audio_system:
            self.audio_system.visualize_system(screen, 20, SCREEN_HEIGHT - 450, 500, 430)
        
        # 小地图
        self._render_minimap()
        
        # 信息
        self._render_info()
        
        pygame.display.flip()
    
    def _render_minimap(self):
        """渲染小地图"""
        map_size = 200
        cell_size = 10
        offset_x = SCREEN_WIDTH - map_size - 20
        offset_y = 20
        
        # 背景
        pygame.draw.rect(screen, (0, 0, 0, 180), 
                        (offset_x - 5, offset_y - 5, map_size + 10, map_size + 10))
        
        # 墙壁
        for wx, wy in self.walls:
            x = offset_x + int(wx * cell_size)
            y = offset_y + int(wy * cell_size)
            color = (150, 150, 150) if wx < 5 or wy < 5 else (120, 120, 120)
            pygame.draw.rect(screen, color, (x, y, cell_size, cell_size))
        
        # 玩家
        px = offset_x + int(self.player_x * cell_size)
        py = offset_y + int(self.player_y * cell_size)
        pygame.draw.circle(screen, (255, 255, 0), (px, py), 6)
        
        # 方向
        dir_len = 15
        dir_x = px + math.cos(self.player_angle) * dir_len
        dir_y = py + math.sin(self.player_angle) * dir_len
        pygame.draw.line(screen, (255, 200, 0), (px, py), (int(dir_x), int(dir_y)), 2)
        
        # 声纳射线
        if self.audio_enabled:
            ray_len = 8 * cell_size
            ray_x = px + math.cos(self.player_angle) * ray_len
            ray_y = py + math.sin(self.player_angle) * ray_len
            pygame.draw.line(screen, (100, 200, 255, 150), 
                           (px, py), (int(ray_x), int(ray_y)), 1)
    
    def _render_info(self):

    avg_distance = sum(distances) / len(distances)

物理原理详解
运动触发机制
if 玩家速度 < 阈值:
    音频信号 = 0
else:
    音频信号 = f(速度, 距离差, 环境密度)

环境密度计算
环境密度 = 1 / (1 + 平均距离到墙壁)
空旷区域 → 平均距离大 → 密度低 → 信号弱
拥挤区域 → 平均距离小 → 密度高 → 信号强

双射线声纳物理
射线1: 从位置P发射
射线2: 从位置P + v·Δt发射 (v是速度向量)
音频信号 ∝ |距离1 - 距离2| × 速度 × 环境密度
